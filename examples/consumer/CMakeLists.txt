cmake_minimum_required(VERSION 3.24)
project(MeshvaleRepairInstalledConsumer LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(MeshvaleRepair CONFIG REQUIRED)
add_executable(meshvale_repair_consumer main.cpp)
target_link_libraries(meshvale_repair_consumer PRIVATE meshvale::repair)
if(MSVC)
  target_compile_options(meshvale_repair_consumer PRIVATE /W4 /WX)
else()
  target_compile_options(meshvale_repair_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
option(MESHVALE_REPAIR_TEST_OFF_FILE_WORKFLOW
  "Exercise the installed reported OFF file workflow development target" OFF)
option(MESHVALE_REPAIR_TEST_OFF_SURFACE_EXPORT
  "Exercise the installed OFF to annotated GLB development target" OFF)
if(MESHVALE_REPAIR_TEST_OFF_SURFACE_EXPORT)
  if(NOT TARGET meshvale::repair_off_surface_export)
    message(FATAL_ERROR "OFF surface consumer requires an installed Repair package built with MESHVALE_REPAIR_OFF_SURFACE_EXPORT=ON")
  endif()
  add_executable(meshvale_off_surface_export_consumer
    ../../tests/off_surface_export_consumer.cpp)
  target_include_directories(meshvale_off_surface_export_consumer PRIVATE ../../tests)
  target_link_libraries(meshvale_off_surface_export_consumer PRIVATE
    meshvale::repair_off_surface_export)
  if(MSVC)
    target_compile_options(meshvale_off_surface_export_consumer PRIVATE /W4 /WX /fp:strict)
  else()
    target_compile_options(meshvale_off_surface_export_consumer PRIVATE
      -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
  endif()
endif()
if(MESHVALE_REPAIR_TEST_OFF_FILE_WORKFLOW)
  if(NOT TARGET meshvale::off_file_workflow)
    message(FATAL_ERROR "OFF workflow consumer requires an installed Repair package built with MESHVALE_REPAIR_OFF_FILE_WORKFLOW=ON")
  endif()
  add_executable(meshvale_off_workflow_consumer off_workflow.cpp)
  target_link_libraries(meshvale_off_workflow_consumer PRIVATE meshvale::off_file_workflow)
  if(MSVC)
    target_compile_options(meshvale_off_workflow_consumer PRIVATE /W4 /WX /fp:strict)
  else()
    target_compile_options(meshvale_off_workflow_consumer PRIVATE
      -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
  endif()
endif()
option(MESHVALE_REPAIR_TEST_GLTF_FILE_WORKFLOW
  "Exercise the installed reported glTF file workflow development target" OFF)
option(MESHVALE_REPAIR_TEST_GLTF_WORKFLOW
  "Exercise the installed reported glTF workflow development target" OFF)
if(MESHVALE_REPAIR_TEST_GLTF_WORKFLOW)
  if(NOT TARGET meshvale::gltf_workflow)
    message(FATAL_ERROR "Workflow consumer requires an installed Repair package built with MESHVALE_REPAIR_GLTF_WORKFLOW=ON")
  endif()
  add_executable(meshvale_gltf_workflow_consumer gltf_workflow.cpp)
  target_link_libraries(meshvale_gltf_workflow_consumer PRIVATE meshvale::gltf_workflow)
  if(MSVC)
    target_compile_options(meshvale_gltf_workflow_consumer PRIVATE /W4 /WX /fp:strict)
  else()
    target_compile_options(meshvale_gltf_workflow_consumer PRIVATE
      -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
  endif()
endif()
if(MESHVALE_REPAIR_TEST_GLTF_FILE_WORKFLOW)
  if(NOT TARGET meshvale::gltf_file_workflow)
    message(FATAL_ERROR "File workflow consumer requires an installed Repair package built with MESHVALE_REPAIR_GLTF_FILE_WORKFLOW=ON")
  endif()
  add_executable(meshvale_gltf_file_workflow_consumer gltf_file_workflow.cpp)
  target_link_libraries(meshvale_gltf_file_workflow_consumer PRIVATE meshvale::gltf_file_workflow)
  if(MSVC)
    target_compile_options(meshvale_gltf_file_workflow_consumer PRIVATE /W4 /WX /fp:strict)
  else()
    target_compile_options(meshvale_gltf_file_workflow_consumer PRIVATE
      -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
  endif()
endif()
add_test(NAME installed_repair_consumer COMMAND meshvale_repair_consumer)
if(TARGET meshvale::materialized_duplicates)
  set(MESHVALE_REPAIR_MATERIALIZED_TEST_PROFILE "qualified" CACHE STRING
    "Numerical producer profile exercised by the charged repair consumer")
  set_property(CACHE MESHVALE_REPAIR_MATERIALIZED_TEST_PROFILE PROPERTY
    STRINGS qualified unsupported)
  if(NOT MESHVALE_REPAIR_MATERIALIZED_TEST_PROFILE STREQUAL "qualified"
      AND NOT MESHVALE_REPAIR_MATERIALIZED_TEST_PROFILE STREQUAL "unsupported")
    message(FATAL_ERROR "MESHVALE_REPAIR_MATERIALIZED_TEST_PROFILE must be qualified or unsupported")
  endif()
  add_executable(meshvale_materialized_duplicate_consumer materialized_duplicates.cpp)
  target_link_libraries(meshvale_materialized_duplicate_consumer PRIVATE meshvale::materialized_duplicates)
  if(MSVC)
    target_compile_options(meshvale_materialized_duplicate_consumer PRIVATE /W4 /WX /fp:strict)
  else()
    target_compile_options(meshvale_materialized_duplicate_consumer PRIVATE
      -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
  endif()
  if(MESHVALE_REPAIR_MATERIALIZED_TEST_PROFILE STREQUAL "qualified")
    add_test(NAME installed_materialized_duplicates COMMAND meshvale_materialized_duplicate_consumer)
  else()
    add_test(NAME installed_materialized_duplicates_unsupported
      COMMAND meshvale_materialized_duplicate_consumer --empty-source)
  endif()
endif()
if(MESHVALE_REPAIR_TEST_GLTF_WORKFLOW)
  if(MESHVALE_REPAIR_MATERIALIZED_TEST_PROFILE STREQUAL "qualified")
    add_test(NAME installed_gltf_workflow COMMAND meshvale_gltf_workflow_consumer)
  else()
    add_test(NAME installed_gltf_workflow_unsupported
      COMMAND meshvale_gltf_workflow_consumer --unsupported-profile)
  endif()
endif()
if(MESHVALE_REPAIR_TEST_GLTF_FILE_WORKFLOW)
  if(MESHVALE_REPAIR_MATERIALIZED_TEST_PROFILE STREQUAL "qualified")
    add_test(NAME installed_gltf_file_workflow COMMAND meshvale_gltf_file_workflow_consumer)
  else()
    add_test(NAME installed_gltf_file_workflow_unsupported
      COMMAND meshvale_gltf_file_workflow_consumer --unsupported-profile)
  endif()
endif()
if(MESHVALE_REPAIR_TEST_OFF_FILE_WORKFLOW)
  if(MESHVALE_REPAIR_MATERIALIZED_TEST_PROFILE STREQUAL "qualified")
    add_test(NAME installed_off_workflow COMMAND meshvale_off_workflow_consumer)
  else()
    add_test(NAME installed_off_workflow
      COMMAND meshvale_off_workflow_consumer --unsupported-profile)
  endif()
endif()
if(MESHVALE_REPAIR_TEST_OFF_SURFACE_EXPORT)
  if(MESHVALE_REPAIR_MATERIALIZED_TEST_PROFILE STREQUAL "qualified")
    add_test(NAME installed_off_surface_export
      COMMAND meshvale_off_surface_export_consumer)
